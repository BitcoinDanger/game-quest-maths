<script>
  // ── STATE ──
  let difficulty = 'easy';
  let score = 0;
  let correct = 0;
  let wrong = 0;
  let qIndex = 0;
  let timerVal = 20;
  let timerInterval = null;
  let answered = false;
  const TOTAL_Q = 10;
  const TIME_PER_Q = 20;

  const ops = ['+', '−', '×', '÷'];

  const ranges = {
    easy:   { max: 10, divMax: 5 },
    medium: { max: 20, divMax: 10 },
    hard:   { max: 50, divMax: 12 }
  };

  function selectDiff(el, diff) {
    document.querySelectorAll('.diff-btn').forEach(b => b.classList.remove('selected'));
    el.classList.add('selected');
    difficulty = diff;
  }

  function rand(min, max) {
    return Math.floor(Math.random() * (max - min + 1)) + min;
  }

  function generateQuestion() {
    const { max, divMax } = ranges[difficulty];
    const op = ops[rand(0, 3)];
    let a, b, answer;

    if (op === '+') {
      a = rand(1, max); b = rand(1, max); answer = a + b;
    } else if (op === '−') {
      a = rand(1, max); b = rand(1, a); answer = a - b;
    } else if (op === '×') {
      a = rand(1, divMax); b = rand(1, divMax); answer = a * b;
    } else {
      b = rand(1, divMax); answer = rand(1, divMax); a = b * answer;
    }

    return { op, a, b, answer };
  }

  function generateChoices(answer) {
    const choices = new Set([answer]);
    while (choices.size < 4) {
      const offset = rand(-15, 15);
      const candidate = answer + offset;
      if (candidate !== answer && candidate >= 0) choices.add(candidate);
    }
    return [...choices].sort(() => Math.random() - 0.5);
  }

  function opLabel(op) {
    const map = { '+': 'Addition', '−': 'Subtraction', '×': 'Multiplication', '÷': 'Division' };
    return map[op];
  }

  function opClass(op) {
    const map = { '+': 'op-add', '−': 'op-sub', '×': 'op-mul', '÷': 'op-div' };
    return map[op];
  }

  // current question data
  let currentAnswer = null;

  function loadQuestion() {
    answered = false;
    const q = generateQuestion();
    currentAnswer = q.answer;

    // update HUD
    document.getElementById('qNumDisplay').textContent = qIndex + 1;
    document.getElementById('progressFill').style.width = ((qIndex / TOTAL_Q) * 100) + '%';
    document.getElementById('feedbackMsg').textContent = '';
    document.getElementById('feedbackMsg').className = 'feedback';

    // op badge
    const badge = document.getElementById('opBadge');
    badge.textContent = opLabel(q.op);
    badge.className = 'op-badge ' + opClass(q.op);

    // question
    document.getElementById('questionText').textContent = `${q.a} ${q.op} ${q.b} = ?`;

    // answers
    const grid = document.getElementById('answersGrid');
    grid.innerHTML = '';
    generateChoices(q.answer).forEach(val => {
      const btn = document.createElement('button');
      btn.className = 'answer-btn';
      btn.textContent = val;
      btn.onclick = () => selectAnswer(btn, val);
      grid.appendChild(btn);
    });

    // timer
    clearInterval(timerInterval);
    timerVal = TIME_PER_Q;
    updateTimer();
    timerInterval = setInterval(() => {
      timerVal--;
      updateTimer();
      if (timerVal <= 0) {
        clearInterval(timerInterval);
        handleTimeout();
      }
    }, 1000);
  }

  function updateTimer() {
    const el = document.getElementById('timerDisplay');
    el.textContent = `⏱ ${timerVal}`;
    el.className = timerVal <= 5 ? 'timer-pill urgent' : 'timer-pill';
  }

  function selectAnswer(btn, val) {
    if (answered) return;
    answered = true;
    clearInterval(timerInterval);
    document.querySelectorAll('.answer-btn').forEach(b => b.disabled = true);

    const fb = document.getElementById('feedbackMsg');
    if (val === currentAnswer) {
      btn.classList.add('correct');
      const bonus = Math.ceil(timerVal / 4);
      score += 10 + bonus;
      correct++;
      fb.textContent = `✅ Correct! +${10 + bonus} pts`;
      fb.className = 'feedback correct-msg';
      document.getElementById('scoreDisplay').textContent = score;
    } else {
      btn.classList.add('wrong');
      wrong++;
      // show correct
      document.querySelectorAll('.answer-btn').forEach(b => {
        if (parseInt(b.textContent) === currentAnswer) b.classList.add('correct');
      });
      fb.textContent = `❌ Answer was ${currentAnswer}`;
      fb.className = 'feedback wrong-msg';
    }

    setTimeout(() => {
      qIndex++;
      if (qIndex >= TOTAL_Q) showResults();
      else loadQuestion();
    }, 1200);
  }

  function handleTimeout() {
    if (answered) return;
    answered = true;
    wrong++;
    document.querySelectorAll('.answer-btn').forEach(b => {
      b.disabled = true;
      if (parseInt(b.textContent) === currentAnswer) b.classList.add('correct');
    });
    const fb = document.getElementById('feedbackMsg');
    fb.textContent = `⏰ Time's up! Answer was ${currentAnswer}`;
    fb.className = 'feedback wrong-msg';
    setTimeout(() => {
      qIndex++;
      if (qIndex >= TOTAL_Q) showResults();
      else loadQuestion();
    }, 1400);
  }

  function startGame() {
    score = 0; correct = 0; wrong = 0; qIndex = 0;
    document.getElementById('scoreDisplay').textContent = 0;
    showScreen('quizScreen');
    loadQuestion();
  }

  function showResults() {
    clearInterval(timerInterval);
    document.getElementById('progressFill').style.width = '100%';

    const pct = correct / TOTAL_Q;
    let emoji, title, sub, stars;
    if (pct === 1)        { emoji='🏆'; title='PERFECT SCORE!'; sub='You are a Math Legend! 🌟'; stars='⭐⭐⭐'; }
    else if (pct >= 0.8)  { emoji='🚀'; title='Stellar Work!'; sub='Almost perfect — keep it up!'; stars='⭐⭐⭐'; }
    else if (pct >= 0.6)  { emoji='🌙'; title='Good Job!'; sub='You\'re getting stronger!'; stars='⭐⭐'; }
    else if (pct >= 0.4)  { emoji='🛸'; title='Keep Trying!'; sub='Practice makes perfect!'; stars='⭐'; }
    else                  { emoji='🌍'; title='Don\'t Give Up!'; sub='Every astronaut starts somewhere!'; stars='⭐'; }

    document.getElementById('resultEmoji').textContent = emoji;
    document.getElementById('resultTitle').textContent = title;
    document.getElementById('resultSub').textContent = sub;
    document.getElementById('starsRow').textContent = stars;
    document.getElementById('finalScore').textContent = score;
    document.getElementById('finalCorrect').textContent = correct;
    document.getElementById('finalWrong').textContent = wrong;

    showScreen('resultsScreen');
    if (pct >= 0.6) launchConfetti();
  }

  function goHome() {
    clearInterval(timerInterval);
    showScreen('homeScreen');
  }

  function showScreen(id) {
    document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
    document.getElementById(id).classList.add('active');
  }

  function launchConfetti() {
    const colors = ['#f7c948','#ff6b6b','#4ecdc4','#a29bfe','#55efc4','#fd79a8'];
    for (let i = 0; i < 60; i++) {
      setTimeout(() => {
        const el = document.createElement('div');
        el.className = 'confetti-piece';
        el.style.left = rand(0, 100) + 'vw';
        el.style.background = colors[rand(0, colors.length - 1)];
        el.style.width = rand(8, 14) + 'px';
        el.style.height = rand(8, 14) + 'px';
